
# ======================================================================
# FILE: assets/.htaccess
# ======================================================================
# Il browser deve sempre ricontrollare JS e CSS del configuratore (dopo un aggiornamento usa i file nuovi)
<IfModule mod_headers.c>
	<FilesMatch "\.(js|mjs|css)$">
		Header set Cache-Control "no-cache"
	</FilesMatch>
</IfModule>

# Riconoscimento del volto: il motore WebAssembly deve arrivare col tipo giusto
<IfModule mod_mime.c>
	AddType application/wasm .wasm
	AddType application/octet-stream .task
</IfModule>

# Nessun file 3D scaricabile dalla cartella del plugin (i pezzi della lampada passano da un endpoint protetto)
<FilesMatch "\.(stl|STL|3mf|flm|obj|step|stp|f3d|blend)$">
	<IfModule mod_authz_core.c>
		Require all denied
	</IfModule>
	<IfModule !mod_authz_core.c>
		Deny from all
	</IfModule>
</FilesMatch>
